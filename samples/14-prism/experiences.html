// a floating window that follows your gaze
import { Space, Window } from '@prism/spatial'

const room = await Space.current()
room.pin(new Window({
  title: 'Aurora Fields',
  size: [1.2, 0.7], // metres
  material: 'glass.iridescent',
  anchor: room.wall('north'),
  follow: { gaze: true, lag: 0.4 }
}))

// pinch to expand, look away to fade
room.on('pinch', w => w.expand({ to: 'volume' }))
room.on('gaze:leave', w => w.fade(0.35))
